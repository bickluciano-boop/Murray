// Pantalla "Mapa" del panel: todos los dispositivos en un mapa, el recorrido y
// las paradas del elegido, y una tarjeta con sus datos y acciones.
//
// Está armada en tres piezas para poder crecer sin rehacerla:
//   1. DATOS: lee lo que el panel ya usa (/api/remote/session y
//      /api/remote/location-history). Cuando exista el modo en vivo o
//      Familia y amigos, solo cambia esta pieza.
//   2. PROVEEDOR DE MAPA: hoy Google Maps. Todo lo que toca google.maps vive
//      en proveedorGoogle(); para cambiar de proveedor (o sumar funciones
//      pagas de Google) se reemplaza esa función y el resto queda igual.
//   3. VISTA: lista, marcadores, tarjeta, recorrido y paradas.
//
// No depende de app-v9.js: se engancha con su propio botón en la barra de
// abajo y se dibuja encima del panel.

const CONFIG = {
  refrescoMs: 15000,        // cada cuánto se piden las ubicaciones con el mapa abierto
  refrescoVivoMs: 8000,     // más seguido si algún dispositivo está "en vivo"
  vivoMs: 2 * 60_000,       // una ubicación de hace menos de esto cuenta como "en vivo"
  puntosRecorrido: 200,     // tope del servidor por pedido
  paradaMetros: 120,        // radio para considerar que se quedó en el mismo lugar
  paradaMinutos: 8,         // tiempo mínimo para marcar una parada
  saltoMaximoMetros: 5000,  // saltos mayores entre dos puntos no suman kilómetros (GPS errático)
  zona: "America/Argentina/Buenos_Aires",
  zoomMaximo: 16,           // al encuadrar, no acercar más que esto (unas cuadras a la vista)
  cercaMetros: 10,          // a esta distancia o menos de quien mira, dice "Cerca" en vez de metros
};

async function pedir(ruta, opciones = {}) {
  const control = new AbortController();
  const corte = setTimeout(() => control.abort(), 20_000);
  try {
    const respuesta = await fetch(ruta, { credentials: "include", cache: "no-store", ...opciones, signal: control.signal });
    const cuerpo = await respuesta.json().catch(() => ({}));
    if (!respuesta.ok) throw Object.assign(new Error(cuerpo.error || "error"), { status: respuesta.status });
    return cuerpo;
  } finally {
    clearTimeout(corte);
  }
}

function numero(valor) {
  const n = Number(valor);
  return Number.isFinite(n) ? n : null;
}

/** Convierte lo que manda el servidor (teléfonos y pulseras) a un formato único. */
function normalizarEquipos(cuenta) {
  const equipos = [];
  for (const d of cuenta.devices || []) {
    if (d.approvalStatus && d.approvalStatus !== "approved") continue;
    const ubic = d.lastLocation || {};
    equipos.push({
      id: String(d.id),
      tipo: "telefono",
      nombre: d.name || "Teléfono",
      plataforma: d.platform || "",
      lat: numero(ubic.latitude),
      lng: numero(ubic.longitude),
      precision: numero(d.accuracy ?? ubic.accuracy),
      direccion: direccionCorta(ubic.address),
      bateria: numero(d.battery),
      enLinea: Boolean(d.online),
      ubicadoEn: d.lastLocationAt || ubic.capturedAt || null,
      vistoEn: d.lastSeenAt || null,
      conRecorrido: true,
    });
  }
  for (const w of cuenta.watches || []) {
    const ubic = w.lastLocation || {};
    equipos.push({
      id: `pulsera:${w.id}`,
      tipo: "pulsera",
      nombre: w.name || "Pulsera",
      plataforma: "Pulsera SOS",
      lat: numero(ubic.latitude),
      lng: numero(ubic.longitude),
      precision: numero(ubic.accuracy),
      direccion: direccionCorta(ubic.address),
      bateria: numero(w.battery),
      enLinea: Boolean(w.online),
      ubicadoEn: w.lastLocationAt || ubic.capturedAt || null,
      vistoEn: w.lastSeenAt || null,
      conRecorrido: false,
    });
  }
  return equipos;
}

async function cargarEquipos() {
  if (DEMO) return demoEquipos();
  const cuenta = await pedir("/api/remote/session");
  if (!cuenta.authenticated) throw Object.assign(new Error("sin_sesion"), { status: 401 });
  return normalizarEquipos(cuenta);
}

async function cargarRecorrido(equipo) {
  if (!equipo.conRecorrido) return [];
  if (DEMO) return demoRecorrido(equipo);
  const datos = await pedir(`/api/remote/location-history?deviceId=${encodeURIComponent(equipo.id)}&limit=${CONFIG.puntosRecorrido}`);
  return (datos.points || [])
    .map((p) => ({ lat: numero(p.latitude), lng: numero(p.longitude), precision: numero(p.accuracy), direccion: direccionCorta(p.address), en: p.capturedAt }))
    .filter((p) => p.lat !== null && p.lng !== null && p.en)
    .sort((a, b) => new Date(a.en) - new Date(b.en));
}

}

const FOTO_VERSIONES = "ojo_ms_fotos_equipos";
function versionFoto(id) {
  try { return JSON.parse(localStorage.getItem(FOTO_VERSIONES) || "{}")[id] || "1"; } catch { return "1"; }
}
function marcarFoto(id) {
  try {
    const versiones = JSON.parse(localStorage.getItem(FOTO_VERSIONES) || "{}");
    versiones[id] = String(Date.now());
    localStorage.setItem(FOTO_VERSIONES, JSON.stringify(versiones));
  } catch { /* navegación privada: la foto nueva aparece al recargar */ }
}
function urlFoto(equipo) {
  if (DEMO || equipo.tipo !== "telefono") return null;
  return `/api/remote/devices/photo?deviceId=${encodeURIComponent(equipo.id)}&v=${versionFoto(equipo.id)}`;
}

/** Achica la foto elegida a 256x256 (recorte al centro) antes de subirla. */
function prepararFoto(archivo) {
  return new Promise((resolver, rechazar) => {
    const img = new Image();
    const url = URL.createObjectURL(archivo);
    img.onload = () => {
      const lado = Math.min(img.naturalWidth, img.naturalHeight);
      const lienzo = document.createElement("canvas");
      lienzo.width = 256; lienzo.height = 256;
      lienzo.getContext("2d").drawImage(img, (img.naturalWidth - lado) / 2, (img.naturalHeight - lado) / 2, lado, lado, 0, 0, 256, 256);
      URL.revokeObjectURL(url);
      resolver(lienzo.toDataURL("image/jpeg", 0.85));
    };
    img.onerror = () => { URL.revokeObjectURL(url); rechazar(new Error("imagen_invalida")); };
    img.src = url;
  });
}

async function subirFoto(equipo, archivo) {
  const image = await prepararFoto(archivo);
  await pedir("/api/remote/devices/photo", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ deviceId: equipo.id, image }) });
  marcarFoto(equipo.id);
}

function metros(a, b) {
  const R = 6371000, rad = Math.PI / 180;
  const dLat = (b.lat - a.lat) * rad, dLng = (b.lng - a.lng) * rad;
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * rad) * Math.cos(b.lat * rad) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(h));
}

function kilometros(puntos) {
  let total = 0;
  for (let i = 1; i < puntos.length; i += 1) {
    const d = metros(puntos[i - 1], puntos[i]);
    if (d < CONFIG.saltoMaximoMetros) total += d;
  }
  return total / 1000;
}

/** Agrupa puntos seguidos que quedan cerca: si duran lo suficiente, es una parada. */
function paradas(puntos) {
  const lista = [];
  let grupo = [];
  const cerrar = () => {
    if (grupo.length < 2) return;
    const desde = new Date(grupo[0].en), hasta = new Date(grupo[grupo.length - 1].en);
    const minutos = (hasta - desde) / 60_000;
    if (minutos < CONFIG.paradaMinutos) return;
    const lat = grupo.reduce((s, p) => s + p.lat, 0) / grupo.length;
    const lng = grupo.reduce((s, p) => s + p.lng, 0) / grupo.length;
    lista.push({ lat, lng, desde, hasta, minutos: Math.round(minutos), direccion: grupo.find((p) => p.direccion)?.direccion || "" });
  };
  for (const punto of puntos) {
    if (!grupo.length) { grupo = [punto]; continue; }
    const centro = { lat: grupo.reduce((s, p) => s + p.lat, 0) / grupo.length, lng: grupo.reduce((s, p) => s + p.lng, 0) / grupo.length };
    if (metros(centro, punto) <= CONFIG.paradaMetros) grupo.push(punto);
    else { cerrar(); grupo = [punto]; }
  }
  cerrar();
  return lista;
}

function inicioDeHoy() {
  const partes = new Intl.DateTimeFormat("en-CA", { timeZone: CONFIG.zona, year: "numeric", month: "2-digit", day: "2-digit" }).format(new Date());
  return new Date(`${partes}T00:00:00-03:00`);
}

function filtrarPorRango(puntos, rango) {
  const desde = rango === "vivo" ? new Date(Date.now() - 2 * 3600_000) : rango === "hoy" ? inicioDeHoy() : new Date(Date.now() - 7 * 86400_000);
  return puntos.filter((p) => new Date(p.en) >= desde);
}

/**
 * "3702, Cerviño, Palermo Botánico, Palermo, Buenos Aires, Comuna 14, ..." -> "Cerviño 3702, Palermo Botánico".
 * Se queda con calle, número y barrio; saca provincia, comuna, código postal y país.
 */
function direccionCorta(texto) {
  const partes = String(texto || "").split(",").map((p) => p.trim()).filter(Boolean);
  if (!partes.length) return "";
  if (/^\d+[a-z]?$/i.test(partes[0]) && partes[1]) partes.splice(0, 2, `${partes[1]} ${partes[0]}`);
  else if (partes[1] && /^\d+[a-z]?$/i.test(partes[1])) partes.splice(0, 2, `${partes[0]} ${partes[1]}`);
  const sobra = /^(argentina|buenos aires|ciudad aut[oó]noma de buenos aires|caba|comuna \d+|[a-z]\d{4}[a-z]{3}|\d{4})$/i;
  const utiles = partes.filter((p, i) => i === 0 || !sobra.test(p));
  const resultado = utiles.slice(0, 2);
  const parecidos = (x, y) => x.toLowerCase().includes(y.toLowerCase()) || y.toLowerCase().includes(x.toLowerCase());
  if (utiles[2] && utiles[1] && !parecidos(utiles[2], utiles[1]) && resultado.join(", ").length < 34) resultado.push(utiles[2]);
  return resultado.join(", ");
}

const horaCorta = (fecha) => new Intl.DateTimeFormat("es-AR", { timeZone: CONFIG.zona, hour: "2-digit", minute: "2-digit", hourCycle: "h23" }).format(new Date(fecha));
const diaCorto = (fecha) => new Intl.DateTimeFormat("es-AR", { timeZone: CONFIG.zona, day: "numeric", month: "short" }).format(new Date(fecha));

function haceCuanto(fecha) {
  if (!fecha) return "sin ubicación";
  const s = Math.max(0, (Date.now() - new Date(fecha).getTime()) / 1000);
  if (s < 60) return `hace ${Math.max(1, Math.round(s))} s`;
  if (s < 3600) return `hace ${Math.round(s / 60)} min`;
  if (s < 86400) return `hace ${Math.round(s / 3600)} h`;
  return `${diaCorto(fecha)} ${horaCorta(fecha)}`;
}

/** Distancia desde donde está quien mira el panel. A 10 m o menos dice "Cerca": el GPS no distingue menos que eso. */
function textoDistancia(m) {
  if (m === null || !Number.isFinite(m)) return "";
  if (m <= CONFIG.cercaMetros) return "Cerca";
  if (m < 100) return `${Math.round(m)} m`;
  if (m < 1000) return `${Math.round(m / 10) * 10} m`;
  if (m < 100_000) return `${(m / 1000).toFixed(1).replace(".", ",")} km`;
  return `${Math.round(m / 1000)} km`;
}

function distanciaA(equipo) {
  if (!estado.yo || !equipo || equipo.lat === null) return null;
  return metros(estado.yo, equipo);
}

function estadoDe(equipo) {
  const edad = equipo.ubicadoEn ? Date.now() - new Date(equipo.ubicadoEn).getTime() : Infinity;
  if (edad <= CONFIG.vivoMs) return { clase: "vivo", texto: "En vivo" };
  if (!equipo.enLinea && edad > 3 * 3600_000) return { clase: "off", texto: "Sin conexión" };
  return { clase: "", texto: "" };
}

function iniciales(nombre) {
  const palabras = String(nombre || "?").split(/\s+/).filter((w) => w && !/^(de|del|la|el|los|las)$/i.test(w));
  return (palabras.length > 1 ? palabras[0][0] + palabras[palabras.length - 1][0] : (palabras[0] || "?").slice(0, 2)).toUpperCase();
}

function avatarHtml(equipo, clase = "") {
  const foto = urlFoto(equipo);
  return `<span class="ogm-av ${clase}"><span>${esc(iniciales(equipo.nombre))}</span>${foto ? `<img src="${foto}" alt="" onerror="this.remove()">` : ""}</span>`;
}

function esperarGoogle(maxMs = 15000) {
  return new Promise((resolver, rechazar) => {
    const inicio = Date.now();
    const mirar = () => {
      if (window.google?.maps?.Map && window.google.maps.OverlayView) return resolver(window.google.maps);
      if (Date.now() - inicio > maxMs) return rechazar(new Error("google_maps_no_cargo"));
      setTimeout(mirar, 200);
    };
    mirar();
  });
}

/**
 * Todo lo que sabe de Google Maps está acá. La vista solo usa estos métodos:
 * marcador, quitarMarcador, recorrido, paradas, precision, encuadrar,
 * centrar, acercar, capa y tema.
 */
async function proveedorGoogle(contenedor, temaInicial) {
  const g = await esperarGoogle();
  class Marcador extends g.OverlayView {
    constructor(posicion, elemento) { super(); this.posicion = posicion; this.elemento = elemento; }
    onAdd() { this.getPanes().overlayMouseTarget.appendChild(this.elemento); g.OverlayView.preventMapHitsAndGesturesFrom?.(this.elemento); }
    draw() {
      const p = this.getProjection()?.fromLatLngToDivPixel(this.posicion);
      if (!p) return;
      this.elemento.style.left = `${p.x}px`;
      this.elemento.style.top = `${p.y}px`;
    }
    onRemove() { this.elemento.remove(); }
    mover(posicion) { this.posicion = posicion; this.draw(); }
  }
  const mapa = new g.Map(contenedor, {
    center: { lat: -34.6037, lng: -58.3816 },
    zoom: 13,
    disableDefaultUI: true,
    gestureHandling: "greedy",
    clickableIcons: false,
    styles: temaInicial === "claro" ? ESTILO_CLARO : ESTILO_OSCURO,
    backgroundColor: temaInicial === "claro" ? "#f2efe9" : "#16181c",
  });
  const marcadores = new Map();
  let lineas = [], marcasParada = [], circulo = null;
  // Overlay vacío: solo sirve para convertir coordenadas a pixeles de pantalla.
  const proyector = new g.OverlayView();
  proyector.onAdd = () => {};
  proyector.draw = () => {};
  proyector.onRemove = () => {};
  proyector.setMap(mapa);
  const pos = (p) => new g.LatLng(p.lat, p.lng);
  return {
    marcador(id, punto, elemento) {
      const actual = marcadores.get(id);
      if (actual && actual.elemento === elemento) { actual.mover(pos(punto)); return; }
      actual?.setMap(null);
      const nuevo = new Marcador(pos(punto), elemento);
      nuevo.setMap(mapa);
      marcadores.set(id, nuevo);
    },
    quitarMarcador(id) { marcadores.get(id)?.setMap(null); marcadores.delete(id); },
    idsMarcadores() { return [...marcadores.keys()]; },
    recorrido(puntos) {
      lineas.forEach((l) => l.setMap(null));
      lineas = [];
      if (puntos.length < 2) return;
      const camino = puntos.map(pos);
      lineas.push(new g.Polyline({ map: mapa, path: camino, strokeColor: "#d9b24c", strokeOpacity: 0.3, strokeWeight: 10, clickable: false }));
      lineas.push(new g.Polyline({ map: mapa, path: camino, strokeColor: "#f0c75a", strokeOpacity: 1, strokeWeight: 3.5, clickable: false }));
    },
    paradas(lista, crearElemento) {
      marcasParada.forEach((m) => m.setMap(null));
      marcasParada = lista.map((parada) => {
        const m = new Marcador(pos(parada), crearElemento(parada));
        m.setMap(mapa);
        return m;
      });
    },
    precision(punto, radio) {
      circulo?.setMap(null);
      circulo = null;
      if (!punto || !radio) return;
      circulo = new g.Circle({ map: mapa, center: pos(punto), radius: radio, fillColor: "#d9b24c", fillOpacity: 0.12, strokeColor: "#d9b24c", strokeOpacity: 0.45, strokeWeight: 1, clickable: false });
    },
    encuadrar(puntos, margen = 80) {
      if (!puntos.length) return;
      if (puntos.length === 1) { mapa.setCenter(pos(puntos[0])); mapa.setZoom(CONFIG.zoomMaximo); return; }
      const limites = new g.LatLngBounds();
      puntos.forEach((p) => limites.extend(pos(p)));
      mapa.fitBounds(limites, margen);
      // Con dispositivos muy juntos, fitBounds acerca hasta ver la vereda: se limita.
      g.event.addListenerOnce(mapa, "idle", () => { if ((mapa.getZoom() || 0) > CONFIG.zoomMaximo) mapa.setZoom(CONFIG.zoomMaximo); });
    },
    centrar(punto, zoom) { mapa.panTo(pos(punto)); if (zoom) mapa.setZoom(zoom); },
    acercar(paso) { mapa.setZoom((mapa.getZoom() || 13) + paso); },
    capa(nombre) { mapa.setMapTypeId(nombre === "satelite" ? "hybrid" : "roadmap"); },
    tema(nombre) { mapa.setOptions({ styles: nombre === "claro" ? ESTILO_CLARO : ESTILO_OSCURO, backgroundColor: nombre === "claro" ? "#f2efe9" : "#16181c" }); },
    redibujar() { g.event.trigger(mapa, "resize"); },
    alTocarMapa(accion) { mapa.addListener("click", accion); },
    alArrastrar(accion) { mapa.addListener("dragstart", accion); },
    alCambiarZoom(accion) { let espera; mapa.addListener("zoom_changed", () => { clearTimeout(espera); espera = setTimeout(accion, 250); }); },
    aPixel(punto) {
      const proy = proyector.getProjection();
      return proy ? proy.fromLatLngToContainerPixel(pos(punto)) : null;
    },
  };
}

  raiz.innerHTML = `

    if (boton.dataset.rango) return elegirRango(boton.dataset.rango);
    if (boton.dataset.capa) {
      raiz.querySelectorAll("[data-capa]").forEach((b) => b.classList.toggle("on", b === boton));
      return estado.proveedor?.capa(boton.dataset.capa);
    }
    if (boton.dataset.id) return elegir(boton.dataset.id, true);
    const accion = boton.dataset.ogm || boton.dataset.accion;
    if (accion === "mas") return estado.proveedor?.acercar(1);
    if (accion === "menos") return estado.proveedor?.acercar(-1);
    if (accion === "centrar") {
      const seguido = estado.vivo && estado.equipos.find((e) => e.id === estado.vivo.id && e.lat !== null);
      if (seguido) { estado.vivo.seguir = true; return estado.proveedor?.centrar(seguido); }
      return encuadrarTodo();
    }
    if (accion === "vivo") return iniciarVivo(equipoElegido());
    if (accion === "vivo-parar") return detenerVivo().then(dibujar);
    if (accion === "foto") return $('[data-ogm="archivo"]', raiz).click();
    if (accion === "llegar") return irConGoogle();
    if (accion === "locate" || accion === "alarm" || accion === "stop-ring") {
      const destino = boton.dataset.equipo ? estado.equipos.find((e) => e.id === boton.dataset.equipo) : equipoElegido();
      return ordenar(accion, boton, destino);
    }
    if (accion === "cerrar-tarjeta") return plegar(!raiz.classList.contains("ogm-plegada"));
  });
  activarArrastre(raiz);
  $('[data-ogm="archivo"]', raiz).addEventListener("change", async (evento) => {
    const archivo = evento.target.files?.[0];
    evento.target.value = "";

    } catch {
      mensajeTarjeta("No se pudo subir la foto. Probá con otra imagen.");
    }
  });
  return raiz;
}

const esCelular = () => window.innerWidth < 900;

/** Tarjeta de abajo en el celular: plegada muestra solo nombre, hora y dirección. */
function plegar(si) {
  $("#og-mapa")?.classList.toggle("ogm-plegada", Boolean(si));
}

/** Deslizar la tarjeta hacia abajo la pliega y hacia arriba la despliega (como en Buscar de Apple). */
function activarArrastre(raiz) {
  const tarjeta = $('[data-ogm="tarjeta"]', raiz);
  let inicioY = null, delta = 0;
  tarjeta.addEventListener("touchstart", (evento) => {
    if (!esCelular()) return;
    const enManija = evento.target.closest(".ogm-asa, .ogm-cab, .ogm-direc");
    if (!enManija && (tarjeta.scrollTop > 0 || evento.target.closest("button, li[data-id]"))) return;
    inicioY = evento.touches[0].clientY;
    delta = 0;
    tarjeta.style.transition = "none";
  }, { passive: true });
  tarjeta.addEventListener("touchmove", (evento) => {
    if (inicioY === null) return;
    delta = evento.touches[0].clientY - inicioY;
    const plegada = raiz.classList.contains("ogm-plegada");
    const mover = plegada ? Math.min(0, delta) * 0.35 : Math.max(0, delta);
    tarjeta.style.transform = `translateY(${mover}px)`;
  }, { passive: true });
  const soltar = () => {
    if (inicioY === null) return;
    tarjeta.style.transition = "";
    tarjeta.style.transform = "";
    if (delta > 45) plegar(true);
    else if (delta < -45) plegar(false);
    inicioY = null;
  };
  tarjeta.addEventListener("touchend", soltar);
  tarjeta.addEventListener("touchcancel", soltar);
}

function agregarPestana() {
  const barra = $(".bottom-tabs");
  if (!barra || $("[data-mapa-tab]", barra)) return;
  const boton = document.createElement("button");
  boton.type = "button";
  boton.className = "bottom-tab";
  boton.setAttribute("data-mapa-tab", "");
  boton.innerHTML = '<span class="tab-icon" aria-hidden="true">◎</span><span>Mapa</span>';
  const inicio = $('[data-view="home"]', barra);
  inicio ? inicio.after(boton) : barra.prepend(boton);
  barra.classList.add("ogm-con-mapa");
  boton.addEventListener("click", () => (estado.abierto ? null : abrir()));
  // Cualquier otra pestaña cierra el mapa (app-v9.js se encarga de mostrar su sección).
  barra.querySelectorAll("[data-view]").forEach((otra) => otra.addEventListener("click", cerrar));
  // app-v9.js vuelve a marcar su pestaña cada vez que refresca el panel (cada 15 s):
  // con el mapa abierto, la única pestaña marcada tiene que ser Mapa.
  new MutationObserver(() => {
    if (!estado.abierto) return;
    barra.querySelectorAll("[data-view].active").forEach((otra) => { otra.classList.remove("active"); otra.setAttribute("aria-current", "false"); });
  }).observe(barra, { subtree: true, attributes: true, attributeFilter: ["class"] });
  document.querySelectorAll("[data-open-view]").forEach((otra) => otra.addEventListener("click", cerrar));
}

async function abrir() {
  const raiz = $("#og-mapa") || construir();
  estado.abierto = true;
  // En el iPhone, si la página quedó desplazada (por ejemplo después de mirar el
  // Historial), Safari achica sus barras y el mapa quedaba corrido: se vuelve arriba
  // y se bloquea el desplazamiento mientras el mapa está abierto.
  estado.desplazamiento = window.scrollY;
  window.scrollTo(0, 0);
  document.documentElement.classList.add("og-mapa-abierto");
  raiz.classList.remove("hidden");
  if (esCelular()) plegar(true);
  document.body.classList.add("og-mapa-abierto");
  document.querySelectorAll(".bottom-tabs .bottom-tab").forEach((b) => {
    const esMapa = b.hasAttribute("data-mapa-tab");
    b.classList.toggle("active", esMapa);
    b.setAttribute("aria-current", esMapa ? "page" : "false");
  });
  if (!estado.proveedor) {
    try {
      estado.proveedor = await proveedorGoogle($('[data-ogm="lienzo"]', raiz), temaActual());
      estado.proveedor.alTocarMapa(() => { if (esCelular()) plegar(true); });
      // Si la persona mueve el mapa a mano, se deja de centrar en el teléfono seguido.
      estado.proveedor.alArrastrar(() => { if (estado.vivo) estado.vivo.seguir = false; });
      estado.proveedor.alCambiarZoom(() => dibujar());
    } catch {
      aviso("No se pudo cargar Google Maps. Revisá la conexión y volvé a abrir el mapa.");
    }
  } else {
    estado.proveedor.redibujar();
  }
  seguirMiUbicacion();
  await refrescar(true);
}

/** Ubicación de quien mira el panel (celular o compu): sirve para "a 2,3 km" / "Cerca". */
let redibujoPendiente = null;
function seguirMiUbicacion() {
  if (estado.miReloj !== null || !navigator.geolocation) return;
  estado.miReloj = navigator.geolocation.watchPosition(
    (pos) => {
      estado.yo = { lat: pos.coords.latitude, lng: pos.coords.longitude, precision: pos.coords.accuracy };
      if (!redibujoPendiente) redibujoPendiente = setTimeout(() => { redibujoPendiente = null; dibujar(); }, 800);
    },
    () => { estado.yo = null; },
    { enableHighAccuracy: true, maximumAge: 10_000, timeout: 20_000 },
  );
}
function dejarMiUbicacion() {
  if (estado.miReloj !== null) navigator.geolocation?.clearWatch(estado.miReloj);
  estado.miReloj = null;
}

function cerrar() {
  if (!estado.abierto) return;
  estado.abierto = false;
  dejarMiUbicacion();
  void detenerVivo();
  clearTimeout(estado.temporizador);
  $("#og-mapa")?.classList.add("hidden");
  document.body.classList.remove("og-mapa-abierto");
  document.documentElement.classList.remove("og-mapa-abierto");
  $("[data-mapa-tab]")?.classList.remove("active");
}

function aviso(texto) {
  const caja = $('[data-ogm="aviso"]');
  if (!caja) return;
  caja.textContent = texto || "";
  caja.classList.toggle("hidden", !texto);
}

async function refrescar(forzarRecorrido = false) {
  clearTimeout(estado.temporizador);
  if (!estado.abierto) return;
  try {
    estado.equipos = await cargarEquipos();
    aviso("");
  } catch (error) {
    if (error.status === 401) aviso("La sesión venció. Volvé a entrar al panel.");
    else aviso("No se pudieron actualizar las ubicaciones. Se reintenta solo.");
  }
  const conUbicacion = estado.equipos.filter((e) => e.lat !== null);
  if (!estado.elegido || !estado.equipos.some((e) => e.id === estado.elegido)) {
    estado.elegido = (conUbicacion[0] || estado.equipos[0])?.id || null;
    forzarRecorrido = true;
  }
  if (!conUbicacion.length && estado.equipos.length) aviso("Todavía no hay ubicaciones. Tocá «Ubicar ahora» para pedir la primera.");
  if (!estado.equipos.length) aviso("Todavía no hay dispositivos vinculados a esta cuenta.");
  const equipo = equipoElegido();
  if (equipo && (forzarRecorrido || Date.now() - estado.ultimoRecorrido > (estado.vivo ? 10_000 : 30_000))) {
    try { estado.recorrido = await cargarRecorrido(equipo); } catch { estado.recorrido = []; }
    estado.ultimoRecorrido = Date.now();
  }
  dibujar();
  if (!estado.encuadrado && conUbicacion.length) { encuadrarTodo(); estado.encuadrado = true; }
  const seguido = estado.vivo?.seguir && estado.equipos.find((e) => e.id === estado.vivo.id && e.lat !== null);
  if (seguido) estado.proveedor?.centrar(seguido);
  const hayVivo = estado.equipos.some((e) => estadoDe(e).clase === "vivo");
  const espera = estado.vivo ? VIVO.refrescoMs : hayVivo ? CONFIG.refrescoVivoMs : CONFIG.refrescoMs;
  estado.temporizador = setTimeout(() => refrescar(), espera);
}

const equipoElegido = () => estado.equipos.find((e) => e.id === estado.elegido) || null;

async function elegir(id, centrar) {
  if (estado.elegido !== id) {
    if (estado.vivo && estado.vivo.id !== id) await detenerVivo();
    estado.mensajeVivo = "";
    estado.mensajeOrden = null;
    estado.elegido = id;
    estado.recorrido = [];
    $("#og-mapa")?.classList.remove("ogm-plegada");
    dibujar();
    try { estado.recorrido = await cargarRecorrido(equipoElegido()); } catch { estado.recorrido = []; }
    estado.ultimoRecorrido = Date.now();
  }
  dibujar();
  const equipo = equipoElegido();
  if (centrar && equipo?.lat !== null) estado.proveedor?.centrar(equipo, CONFIG.zoomMaximo);
}

function elegirRango(rango) {
  estado.rango = rango;
  document.querySelectorAll("#og-mapa [data-rango]").forEach((b) => b.classList.toggle("on", b.dataset.rango === rango));
  dibujar();
  const puntos = filtrarPorRango(estado.recorrido, rango);
  if (puntos.length > 1) estado.proveedor?.encuadrar(puntos, margenMapa());
}

/**
 * Dispositivos casi en el mismo lugar (por ejemplo el teléfono y la notebook en casa)
 * se tapaban entre sí: se apilan uno arriba del otro, centrados en el punto real.
 */
function desplazamientos(equipos, aPixel = null, radioPixeles = 52, radioMetros = 60, paso = 84) {
  const resultado = new Map();
  const pendientes = equipos.filter((e) => e.lat !== null);
  // Con el mapa cargado se mide en pantalla (así funciona con cualquier zoom); si no, en metros.
  const cerca = (a, b) => {
    const pa = aPixel?.(a), pb = aPixel?.(b);
    return pa && pb ? Math.hypot(pa.x - pb.x, pa.y - pb.y) <= radioPixeles : metros(a, b) <= radioMetros;
  };
  while (pendientes.length) {
    const base = pendientes.shift();
    const grupo = [base];
    for (let i = pendientes.length - 1; i >= 0; i -= 1) {
      if (cerca(base, pendientes[i])) grupo.push(...pendientes.splice(i, 1));
    }
    grupo.sort((a, b) => a.nombre.localeCompare(b.nombre));
    grupo.forEach((e, i) => resultado.set(e.id, grupo.length > 1 ? Math.round((i - (grupo.length - 1) / 2) * paso) : 0));
  }
  return resultado;
}

function elementoMarcador(equipo, dy = 0) {
  const el = estado.elementosMarcador.get(equipo.id) || document.createElement("button");
  el.type = "button";
  el.dataset.id = equipo.id;
  const st = estadoDe(equipo);
  el.className = `ogm-pin ${st.clase}${equipo.id === estado.elegido ? " sel" : ""}`;
  el.style.marginTop = dy ? `${dy}px` : "";
  el.innerHTML = `<span class="ogm-halo"></span>${avatarHtml(equipo)}<span class="ogm-etq">${esc(equipo.nombre)}<em>${esc(haceCuanto(equipo.ubicadoEn))}</em></span>`;
  if (!estado.elementosMarcador.has(equipo.id)) {
    el.addEventListener("click", (evento) => { evento.stopPropagation(); elegir(equipo.id, false); });
    estado.elementosMarcador.set(equipo.id, el);
  }
  return el;
}

function elementoParada(parada) {
  const el = document.createElement("button");
  el.type = "button";
  el.className = "ogm-parada";
  const texto = `Estuvo ${parada.minutos >= 60 ? `${Math.floor(parada.minutos / 60)} h ${parada.minutos % 60} min` : `${parada.minutos} min`} · ${horaCorta(parada.desde)} a ${horaCorta(parada.hasta)}`;

  el.addEventListener("click", (evento) => { evento.stopPropagation(); el.classList.toggle("abierta"); });
  return el;
}

function dibujar() {
  const raiz = $("#og-mapa");
  if (!raiz) return;
  const equipos = estado.equipos;
  const elegido = equipoElegido();

  // Lista lateral (escritorio) / tira de abajo (celular)
  const conUbic = equipos.filter((e) => e.lat !== null).length;
  $('[data-ogm="resumen"]', raiz).textContent = `${equipos.length} dispositivo${equipos.length === 1 ? "" : "s"} · ${conUbic} con ubicación`;
  $('[data-ogm="lista"]', raiz).innerHTML = equipos.map((e) => {
    const st = estadoDe(e);
    const bat = e.bateria === null ? "" : ` <span class="ogm-bat${e.bateria < 25 ? " baja" : ""}" title="Batería ${Math.round(e.bateria)}%"><i style="width:${Math.max(4, Math.min(100, e.bateria))}%"></i></span>`;
    const dist = textoDistancia(distanciaA(e));
    return `<li data-id="${esc(e.id)}" class="${e.id === estado.elegido ? "sel" : ""}" tabindex="0">${avatarHtml(e, "chico")}<div><span class="ogm-nombre"><strong>${esc(e.nombre)}</strong>${dist ? `<em class="ogm-dist">${esc(dist)}</em>` : ""}</span><small>${st.texto ? `<b class="${st.clase}">${st.clase === "vivo" ? "● " : ""}${esc(st.texto)}</b> · ` : ""}${esc(haceCuanto(e.ubicadoEn))}${bat}</small>${e.direccion ? `<small class="ogm-dir">${esc(e.direccion)}</small>` : ""}</div>${botonSonar(e)}</li>`;
  }).join("");

  // Marcadores
  const p = estado.proveedor;
  if (p) {
    const vigentes = new Set();
    const corrimientos = desplazamientos(equipos, p.aPixel);
    for (const e of equipos) {
      if (e.lat === null) continue;
      vigentes.add(e.id);
      p.marcador(e.id, e, elementoMarcador(e, corrimientos.get(e.id) || 0));
    }
    if (estado.yo) {
      vigentes.add("__yo");
      if (!estado.elementoYo) {

        estado.elementoYo.className = "ogm-yo";

    const trazo = elegido && elegido.lat !== null && puntos.length ? [...puntos, { lat: elegido.lat, lng: elegido.lng }] : puntos;

    <div class="ogm-cab">
      <button type="button" class="ogm-foto" data-accion="foto" title="Cambiar foto" ${elegido.tipo !== "telefono" || DEMO ? "disabled" : ""}>${avatarHtml(elegido, "grande")}</button>

"● " : ""}${esc(st.texto)}</b> · ` : ""}${esc(haceCuanto(elegido.ubicadoEn))}${distanciaA(elegido) !== null ? ` · ${esc(textoDistancia(distanciaA(elegido)))}` : ""}`}</small></div>

    </div>
    ${estado.mensajeVivo ? `<p class="ogm-aviso-vivo">${esc(estado.mensajeVivo)}</p>` : ""}
    <p class="ogm-msj" data-ogm="msj">${esc(estado.mensajeOrden?.id === elegido.id ? estado.mensajeOrden.texto : "")}</p>
    ${otros.length ? `<ul class="ogm-otros">${otros.map((e) => `<li data-id="${esc(e.id)}">${avatarHtml(e, "mini")}<span>${esc(e.nombre)}</span><small>${esc([textoDistancia(distanciaA(e)), haceCuanto(e.ubicadoEn)].filter(Boolean).join(" · "))}</small>${botonSonar(e)}</li>`).join("")}</ul>` : ""}`;
}

// ---------------------------------------------------------------------------
// Seguir en vivo: el teléfono manda su ubicación cada pocos segundos mientras
// el mapa está abierto (ver /api/remote/live y src/liveLocation.ts en la app).
// ---------------------------------------------------------------------------

const VIVO = { renovarMs: 4 * 60_000, refrescoMs: 4_000, vigilarIntentos: 40 };

/** Solo teléfonos con la app: la notebook (programa de la computadora) no tiene seguimiento en vivo. */
function admiteVivo(equipo) {
  return Boolean(equipo) && equipo.tipo === "telefono" && !/windows|mac ?os|linux|computadora|notebook|\bpc\b/i.test(`${equipo.plataforma} ${equipo.nombre}`);
}

function pedirVivo(accion, deviceId) {
  if (DEMO) return Promise.resolve({ live: accion === "start", demo: true });
  return pedir("/api/remote/live", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ deviceId, action: accion }) });
}

async function iniciarVivo(equipo) {
  if (!admiteVivo(equipo) || estado.vivo?.id === equipo.id) return;
  await detenerVivo();
  estado.mensajeVivo = "";
  estado.vivo = { id: equipo.id, seguir: true, texto: "Conectando con el teléfono…", renovador: null };
  dibujar();
  try {
    const r = await pedirVivo("start", equipo.id);
    if (estado.vivo?.id !== equipo.id) return;
    if (r.demo) estado.vivo.texto = "Demostración: no se envía nada a un teléfono real";
    else if (r.pushSent === false) estado.vivo.texto = "Esperando que el teléfono se conecte…";
    estado.vivo.renovador = setInterval(() => {
      if (estado.vivo) pedirVivo("start", estado.vivo.id).catch(() => undefined);
    }, VIVO.renovarMs);
    void vigilarOrdenVivo(equipo.id, r.commandId);
  } catch (error) {
    estado.vivo = null;
    estado.mensajeVivo = error.status === 401 ? "La sesión venció. Volvé a entrar al panel." : "No se pudo activar el seguimiento en vivo. Probá de nuevo.";
  }
  void refrescar(true);
}

/** Mira la orden hasta que el teléfono la toma, para avisar si no pudo arrancar (por ejemplo, falta un permiso). */
async function vigilarOrdenVivo(id, commandId) {
  if (!commandId || DEMO) return;
  for (let intento = 0; intento < VIVO.vigilarIntentos && estado.vivo?.id === id; intento += 1) {
    await new Promise((ok) => setTimeout(ok, 3000));
    const c = (await estadoOrden(commandId).catch(() => null))?.command;
    if (!c || estado.vivo?.id !== id) continue;
    if (c.status === "executed") { estado.vivo.texto = ""; dibujar(); return; }
    if (c.status === "failed") {
      clearInterval(estado.vivo.renovador);
      estado.vivo = null;
      estado.mensajeVivo = c.error === "background_location_not_granted"
        ? "El teléfono necesita el permiso de ubicación «Permitir todo el tiempo» (Ajustes › Apps › Ojo Guard › Ubicación)."
        : `El teléfono no pudo empezar el seguimiento${c.error ? ` (${c.error})` : ""}.`;
      dibujar();
      return;
    }
  }
}

async function detenerVivo() {
  const vivo = estado.vivo;
  if (!vivo) return;
  clearInterval(vivo.renovador);
  estado.vivo = null;
  await pedirVivo("stop", vivo.id).catch(() => undefined);
}

function mensajeTarjeta(texto, id = estado.elegido) {
  // Se guarda para que no se borre cuando la tarjeta se vuelve a dibujar (cada pocos segundos).
  estado.mensajeOrden = id ? { id, texto } : null;
  const caja = $('#og-mapa [data-ogm="msj"]');
  if (caja && id === estado.elegido) caja.textContent = texto;
}

function irConGoogle() {
  const e = equipoElegido();
  if (!e || e.lat === null) return;
  window.open(`https://www.google.com/maps/dir/?api=1&destination=${e.lat},${e.lng}`, "_blank", "noopener");
}

async function ordenar(tipo, boton, equipo = equipoElegido()) {
  if (!equipo) return;
  const avisar = (texto) => mensajeTarjeta(texto, equipo.id);
  const conTexto = !boton.classList.contains("ogm-sonar");
  const etiqueta = boton.textContent;
  boton.disabled = true;
  if (conTexto) boton.textContent = tipo === "locate" ? "Ubicando…" : tipo === "alarm" ? "Haciendo sonar…" : "Deteniendo…";
  avisar("Enviando orden…");
  try {
    const r = await enviarOrden(equipo, tipo);
    if (!r.demo && (!r.queued || !r.commandId)) throw new Error("no_encolada");
    // Ni bien sale la orden de sonar ya se puede detener (no hace falta esperar a que el teléfono confirme).
    if (tipo === "alarm") estado.sonando = { id: equipo.id, hasta: Date.now() + 70_000 };
    if (tipo === "stop-ring") estado.sonando = null;
    if (r.demo) { avisar("Demostración: no se envía ninguna orden a un teléfono real."); dibujar(); return; }
    if (tipo !== "locate") dibujar();
    avisar(r.pushSent ? "Orden enviada · esperando al teléfono." : "Orden guardada · se ejecuta cuando el teléfono se conecte.");
    for (let intento = 0; intento < 20 && estado.abierto; intento += 1) {
      await new Promise((ok) => setTimeout(ok, 3000));
      const c = (await estadoOrden(r.commandId).catch(() => null))?.command;
      if (!c) continue;
      if (c.status === "executed") {
        avisar(tipo === "locate" ? "Ubicación recibida." : tipo === "alarm" ? "El teléfono está sonando. Se corta solo al minuto." : "Sonido detenido.");
        estado.ultimoRecorrido = 0;
        await refrescar(true);
        break;
      }
      if (c.status === "failed") { avisar(c.error ? `No se pudo: ${c.error}` : "El teléfono no pudo completar la orden."); break; }
    }
  } catch (error) {
    avisar(error.status === 403
      ? "Esta acción pide confirmar tu identidad: hacela desde la pestaña Dispositivos."
      : "No se pudo enviar la orden. Probá de nuevo.");
  } finally {
    boton.disabled = false;
    if (conTexto) boton.textContent = etiqueta;
  }
}
